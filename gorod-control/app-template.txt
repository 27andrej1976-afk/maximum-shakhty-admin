  const STREET_PREFIX = /^(?:г\.?\s*шахты\s*,?\s*)?(?:(?:ул(?:ица)?|пер(?:еулок)?|проспект|пр-?кт|пр-?т|пр-?д|пр(?:оезд)?-?д|пл(?:ощадь)?|бул(?:ьвар)?)\.?\s+)/i;
  const clean = (s) => String(s || '').toLowerCase().replace(/ё/g, 'е').replace(/[«»"'`]/g, '').replace(/\s+/g, ' ').trim();
  const kindRules = [
    ['ул', /^(?:улица|ул\.?)\s+(.+)$/i, /^(.+)\s+(?:улица|ул\.?)$/i],
    ['пер', /^(?:переулок|пер\.?)\s+(.+)$/i, /^(.+)\s+(?:переулок|пер\.?)$/i],
    ['просп', /^(?:проспект|пр-?кт|пр-?т|пр\.?)\s+(.+)$/i, /^(.+)\s+(?:проспект|пр-?кт|пр-?т)$/i],
    ['проезд', /^(?:проезд|пр-д)\s+(.+)$/i, /^(.+)\s+проезд$/i],
    ['пл', /^(?:площадь|пл\.?)\s+(.+)$/i, /^(.+)\s+площадь$/i],
    ['бул', /^(?:бульвар|бул\.?)\s+(.+)$/i, /^(.+)\s+бульвар$/i],
    ['пос', /^(?:поселок|посёлок|пос\.?)\s+(.+)$/i, /^(.+)\s+(?:поселок|посёлок)$/i],
    ['мкр', /^(?:микрорайон|мкр\.?)\s+(.+)$/i, /^(.+)\s+микрорайон$/i]
  ];
  function normalStreet(str) {
    const raw=clean(str).replace(/^(?:г\.?\s*шахты)\s*,?\s*/i,'').replace(/\bдом\b.*$/,'').replace(/[.,;]+$/,'').trim();
    if(!raw)return '';
    for(const [kind,lead,tail] of kindRules){
      const m=raw.match(lead)||raw.match(tail);
      if(m)return kind+':'+m[1].replace(/[.,;]+$/,'').trim();
    }
    return 'неуточн:'+raw;
  }
  function normalHouse(str) {
    return clean(str).replace(/^(?:д(?:ом)?\.?\s*)/,'').replace(/\s/g, '').replace(/[^\dа-яa-z\/-]/g, '');
  }
  function parseHomeAddress(input) {
    const text = String(input || '').trim().replace(/^россия\s*,?\s*/i, '').replace(/^(?:ростовская область\s*,?\s*)/i,'').replace(/^(?:г\.?\s*)?шахты\s*,?\s*/i,'');
    const match = text.match(/^(.*?)(?:\s*,\s*(?:д\.?\s*)?|\s+(?:д\.?\s*))([0-9]+\s*[а-яa-z]?)\s*$/i);
    if (match) return { street: clean(match[1]), house: normalHouse(match[2]) };
    const alt = text.match(/^((?:ул\.|улица|пер\.|переулок|проспект|пр-т|пр\.|пр-д|проезд)\s+.+?)\s+([0-9]+[а-яa-z]?)$/i);
    if (alt) return { street: clean(alt[1]), house: normalHouse(alt[2]) };
    return { street: clean(text), house: '' };
  }
  function distanceM(a, b) {
    if (!a || !b || !Number.isFinite(+a.lat) || !Number.isFinite(+a.lon) || !Number.isFinite(+b.lat) || !Number.isFinite(+b.lon)) return Infinity;
    const R = 6371000, rad = x => x * Math.PI / 180;
    const dLat = rad(+b.lat - +a.lat), dLon = rad(+b.lon - +a.lon);
    const x = Math.sin(dLat / 2)**2 + Math.cos(rad(+a.lat)) * Math.cos(rad(+b.lat)) * Math.sin(dLon / 2)**2;
    return 2 * R * Math.atan2(Math.sqrt(x), Math.sqrt(1 - x));
  }
  function houseInRanges(house, ranges) {
    if (!house) return false;
    const hn = Number(String(house).match(/^\d+/)?.[0]);
    if (!Number.isFinite(hn)) return false;
    return (ranges || []).some(v => {
      const t = normalHouse(v);
      if (t === house) return true;
      const m = t.match(/^(\d+)-(\d+)$/);
      if (!m) return false;
      const lo = +m[1], hi = +m[2], min = Math.min(lo, hi), max = Math.max(lo, hi);
      if (hn < min || hn > max) return false;
      return (lo % 2 === hi % 2) ? (hn % 2 === lo % 2) : true;
    });
  }
  function matchImpact(event, home, radiusM=1000) {
    if (!home || !Number.isFinite(+home.lat) || !Number.isFinite(+home.lon)) return { relevant:false, reason:'Нет контрольной точки', confidence:'unknown' };
    if (['resolved','cancelled','canceled'].includes(event.status)) return { relevant:false, reason:'Работы завершены', confidence:'high' };
    if (event.category === 'mchs' && (event.scope === 'city' || event.scope === 'region')) {
      const expiry=Date.parse(event.end||'');
      if(Number.isFinite(expiry)&&expiry<Date.now()-3600000)return {relevant:false,reason:'Период предупреждения завершён',confidence:'high'};
      const saved=clean(home?.label||'');
      if (!home || !Number.isFinite(+home.lat) || !Number.isFinite(+home.lon) || !saved.includes('шахты'))
        return {relevant:false,reason:'Предупреждение относится к Шахтам, но адрес не подтверждён',confidence:'unknown'};
      if(distanceM(home,{lat:47.7086,lon:40.2146})>25000)
        return {relevant:false,reason:'Адрес вне тестовой географии Шахт',confidence:'high'};
      return {relevant:true,reason:event.scope==='city'?'Предупреждение для города Шахты':'Предупреждение для всей Ростовской области',
        confidence:'high',match:'area',distance:NaN};
    }
    const today = Date.now(), end = event.end ? Date.parse(event.end) : NaN, start = event.start ? Date.parse(event.start) : NaN;
    if(['power','water'].includes(event.category) && today>=Date.parse(event.end||event.start||event.publishedAt||event.observedAt||'')+36*3600000)return {relevant:false,reason:'Истёк срок сохранения 36 часов',confidence:'unknown'};
    if (event.category!=='power' && Number.isFinite(end) && end < today - 24*3600000) return { relevant:false, reason:'Истёк срок актуальности сообщения, требуется новая проверка', confidence:'unknown' };
    if (event.kind === 'planned' && Number.isFinite(start) && start > today + 14*86400000) return { relevant:false, reason:'План слишком далеко', confidence:'high' };
    if (event.kind !== 'planned' && !Number.isFinite(end) && Number.isFinite(start) && start < today - 48*3600000) return { relevant:false, reason:'Устаревшее сообщение', confidence:'high' };
    const street = normalStreet(home.street);
    let streetMatch = false, exactMatch = false, possibleMatch = false;
    for (const area of (event.affected || [])) {
      const candidate = normalStreet(area.street);
      if (!street || !candidate || street !== candidate) continue;
      streetMatch = true;
      const homes = Array.isArray(area.houses) ? area.houses : [];
      if (homes.length) {
        if (houseInRanges(normalHouse(home.house), homes)) exactMatch = true;
      } else possibleMatch = true;
    }
    const dist = distanceM(event.point, home);
    if (exactMatch) return { relevant:true, reason:'Ваш дом указан в зоне отключения', confidence:'high', match:'address', distance:dist };
    if (possibleMatch) return { relevant:true, reason:'Указана ваша улица — дом нужно уточнить', confidence:'medium', match:'street', distance:dist };
    if (event.kind === 'planned' && streetMatch && (event.affected || []).length) return { relevant:false, reason:'Ваш номер дома не входит в опубликованный список', confidence:'high', distance:dist };
    if (Number.isFinite(dist) && dist <= radiusM) return { relevant:true, reason:`Место работ приблизительно в ${Math.round(dist)} м`, confidence:'medium', match:'radius', distance:dist };
    return { relevant:false, reason:streetMatch ? 'Дом не входит в указанные номера' : 'За пределами зоны', confidence:'high', distance:dist };
  }
  root.CityLogic = { normalStreet, normalHouse, parseHomeAddress, distanceM, houseInRanges, matchImpact };
})(typeof window !== 'undefined' ? window : globalThis);

</script>

async function checkSources(){
  if(state.demo){toast('Отключите тестовый режим для проверки официальных публикаций');return}
  $('checkButton').disabled=true;$('refreshIcon').textContent='…';
  try{
    const r=await fetch('./data/water.json?_='+Date.now(),
      {headers:{'Accept':'application/json'},cache:'no-store',signal:AbortSignal.timeout(18000)});
    if(!r.ok)throw Error('No published feed');
    const data=await r.json();
    if(data.schema!==2||!Array.isArray(data.sources)||!Array.isArray(data.events))throw Error('Invalid feed');
    state.checkedAt=data.checkedAt||null;
    state.sourceRecent=data.recent||[];
    state.checks=data.sources;
    const scanAge=Date.now()-Date.parse(state.checkedAt||0);
    const scanRecent=Number.isFinite(scanAge)&&scanAge>=0&&scanAge<4*3600000;
    const sourceById=Object.fromEntries(state.checks.map(s=>[s.id,s]));
    const isCurrent=s=>{const age=Date.now()-Date.parse(s?.checkedAt||state.checkedAt||'');
      return !!s&&(s.status==='ok'||s.readStatus==='ok')&&Number.isFinite(age)&&age>=-120000&&age<4*3600000;};
    state.backendConnected=scanRecent&&state.checks.some(isCurrent);
    state.events=scanRecent?data.events.filter(e=>{
      const url=String(e.url||'');
      if(e.category==='water')return waterSourceAccepted(e,sourceById,isCurrent);
      if(e.category==='power'){
        if(e.origin==='official_city_post'&&e.sourceId==='power_city')
          return isCurrent(sourceById.power_city)&&url.startsWith('https://t.me/admshakhty/');
        if(e.origin!=='attributed_repost')return false;
        if(url.startsWith('https://newsrostov.ru/'))return isCurrent(sourceById.power_repost_rostov);
        if(url.startsWith('https://news-don.ru/'))return isCurrent(sourceById.power_repost_don);
        return false;
      }
      if(e.category==='gas'){
        return e.sourceId==='gas_city'&&e.origin==='official_city_post'&&
          url.startsWith('https://t.me/admshakhty/')&&isCurrent(sourceById.gas_city);
      }
      if(e.category==='mchs'){
        return isCurrent(sourceById.mchs)&&e.sourceId==='mchs'&&
          /^https:\/\/61\.mchs\.gov\.ru\/deyatelnost\/press-centr\/operativnaya-informaciya\//.test(url)&&
          (e.scope==='city'||e.scope==='region');
      }
      return false;
    }):[];
    toast(state.backendConnected?
      'Проверены доступные объявления воды, электричества и МЧС. Статус служб приведён ниже.':
      'Не удалось подтвердить свежесть сообщений всех коммунальных служб.');
  }catch(err){
    state.backendConnected=false;state.events=[];state.checks=[];
    toast('Не удалось загрузить городскую сводку. Можно проверить учебные сценарии.');
  }finally{
    state.lastLocalCheck=Date.now();
    $('checkButton').disabled=false;$('refreshIcon').textContent='↻';
    renderSources();renderEvents();
  }
}

async function applyGPSLocation(lat,lon,accuracy){
  const latitude=Number(lat),longitude=Number(lon);
  if(!Number.isFinite(latitude)||!Number.isFinite(longitude)||latitude<47||latitude>49||longitude<39||longitude>42){
    finishGeo('Координаты не относятся к району Шахт. Для контроля дома в Шахтах введите адрес вручную.');
    return;
  }
  setGeoStatus('Координаты получены. Определяем улицу и номер дома…');
  const geo=await reverseGeocode(latitude,longitude);
  saveHome({lat:latitude,lon:longitude,street:geo.street,house:geo.house,label:geo.label,
    accuracy: Number.isFinite(Number(accuracy))?Number(accuracy):null,method:'gps'});
  $('geoButton').disabled=false;
  state.geoPending=false;
  if(!geo.street||!geo.house){
    const message='GPS определил точку, но номер дома не подтверждён. Нажмите «Ввести адрес» и укажите улицу с номером: иначе адресные отключения могут не находиться.';
    setGeoStatus(message);
    toast('Координаты получены. Уточните адрес дома для точных предупреждений.');
  }else if(Number(accuracy)>750){
    setGeoStatus('Координаты получены с низкой точностью (около '+Math.round(Number(accuracy))+' м). Для проверки дома уточните адрес вручную.');
    toast('GPS работает, но точность низкая.');
  }else{
    setGeoStatus('Местоположение определено. Проверьте адрес дома на карте.');
    toast('Геолокация определена. Адрес сохранён на устройстве.');
  }
  if(!state.demo)checkSources();
}

function geoLocate(){
  if(state.geoPending)return;
  state.geoPending=true;
  const attempt=(state.gpsRequestId||0)+1;
  state.gpsRequestId=attempt;
  $('geoButton').disabled=true;
  setGeoStatus('Определяем местоположение. Разрешите доступ к геолокации, когда Android попросит.');
  if(/CityControlApp\//.test(navigator.userAgent)){
    // Android native location: independent of WebView's browser geolocation implementation.
    try{window.location.href='citycontrol://get-location'}
    catch(_){window.CityControlNativeGPSError('UNAVAILABLE')}
    setTimeout(()=>{
      if(state.geoPending&&state.gpsRequestId===attempt)
        window.CityControlNativeGPSError('TIMEOUT');
    },38000);
    return;
  }
  if(!navigator.geolocation){
    finishGeo('Браузер не поддерживает геолокацию. Введите адрес вручную.');
    return;
  }
  navigator.geolocation.getCurrentPosition(
    pos=>applyGPSLocation(pos.coords.latitude,pos.coords.longitude,pos.coords.accuracy).catch(
      ()=>finishGeo('Не удалось сохранить точку, укажите адрес вручную.')),
    error=>{
      if(error.code===1){
        finishGeo('Браузеру запрещено определять местоположение. Разрешите геолокацию для этого сайта или укажите адрес вручную.');
      }else if(error.code===3){
        navigator.geolocation.getCurrentPosition(
          pos=>applyGPSLocation(pos.coords.latitude,pos.coords.longitude,pos.coords.accuracy).catch(
            ()=>finishGeo('Не удалось сохранить точку, укажите адрес вручную.')),
          ()=>finishGeo('Не удалось получить координаты. Проверьте GPS или введите адрес вручную.'),
          {enableHighAccuracy:false,timeout:18000,maximumAge:180000});
      }else finishGeo('Не удалось определить местоположение. Проверьте GPS или введите адрес вручную.');
    },
    {enableHighAccuracy:true,timeout:18000,maximumAge:120000});
}
